-- SLIP 6 : Factorial of a number
------------------------------------------------------------
DECLARE
  n NUMBER := &n;
  f NUMBER := 1;
BEGIN
  FOR i IN 1..n LOOP
    f := f * i;
  END LOOP;
  DBMS_OUTPUT.PUT_LINE('Factorial of ' || n || ' is ' || f);
END;

-- SLIP 8 : Cursor - employees of computer department
------------------------------------------------------------
DECLARE
  CURSOR c_emp IS
    SELECT e.empno, e.ename, e.job, e.sal
    FROM emp e, dept d
    WHERE e.deptno = d.deptno AND d.dname = 'COMPUTER';

    DBMS_OUTPUT.PUT_LINE(r.empno || ' ' || r.ename || ' ' || r.job || ' ' || r.sal);
  END LOOP;
END;

-- SLIP 9 : Predefined exception (ZERO_DIVIDE)
------------------------------------------------------------

END;

------------------------------------------------------------
-- SLIP 20 : Factorial function (function + call in ONE block)
------------------------------------------------------------
DECLARE
  num NUMBER := &num;

  FUNCTION fact (n NUMBER) RETURN NUMBER IS
    f NUMBER := 1;
  BEGIN
    FOR i IN 1..n LOOP
      f := f * i;
    END LOOP;
    RETURN f;
  END;
BEGIN
  DBMS_OUTPUT.PUT_LINE('Factorial of ' || num || ' is ' || fact(num));
END;
/

------------------------------------------------------------
-- SLIP 21 : Function - maximum of two values (function + call in ONE block)
------------------------------------------------------------
DECLARE
  x NUMBER := &x;
  y NUMBER := &y;

  FUNCTION max_of_two (a NUMBER, b NUMBER) RETURN NUMBER IS
  BEGIN
    IF a >= b THEN
      RETURN a;
    ELSE
      RETURN b;
    END IF;
  END;
BEGIN
  DBMS_OUTPUT.PUT_LINE('Maximum is ' || max_of_two(x, y));
END;
/

------------------------------------------------------------
-- SLIP 22 : Factorial function (function + call in ONE block)
------------------------------------------------------------
DECLARE
  num NUMBER := &num;

  FUNCTION factorial (n NUMBER) RETURN NUMBER IS
    f NUMBER := 1;
    i NUMBER := n;
  BEGIN
    WHILE i > 1 LOOP
      f := f * i;
      i := i - 1;
    END LOOP;
    RETURN f;
  END;
BEGIN
  DBMS_OUTPUT.PUT_LINE('Factorial of ' || num || ' is ' || factorial(num));
END;
/
